}

        <a class="service-card__link" href="#contact">${I18N[state.language].services.link} →</a>
      </div>
    </article>
  ).join("");
}

function renderSteps() {
  const steps = $("[data-steps]");
  steps.innerHTML = I18N[state.language].http://process.steps.map((step, index) => 
    <article class="step">
      <span class="step__num">${index + 1}</span>
      <h3>${step.title}</h3>
      <p>${step.text}</p>
    </article>
  ).join("");
}

function initLanguage() {
  const selector = $("[data-language]");
  selector.addEventListener("change", (event) => {
    state.language = http://event.target.value;
    document.documentElement.lang = state.language;
    applyStaticText();
    renderServices();
    renderSteps();
    initReveal();
  });
}

function initNavigation() {
  const toggle = $("[data-nav-toggle]");
  const menu = $("[data-nav-menu]");
  toggle.addEventListener("click", () => {
    const open = !menu.classList.contains("is-open");
    menu.classList.toggle("is-open", open);
    document.body.classList.toggle("nav-open", open);
    toggle.setAttribute("aria-expanded", String(open));
  });

  $$("a[href^='#']", menu).forEach((link) => {
    link.addEventListener("click", () => {
      menu.classList.remove("is-open");
      document.body.classList.remove("nav-open");
      toggle.setAttribute("aria-expanded", "false");
    });
  });

  const header = $("[data-header]");
  const onScroll = () => header.classList.toggle("is-scrolled", window.scrollY > 12);
  onScroll();
  window.addEventListener("scroll", onScroll, { passive: true });
}

function initReveal() {
  const items = $$(".reveal:not(.is-visible)");
  if (!("IntersectionObserver" in window)) {
    items.forEach((item) => item.classList.add("is-visible"));
    return;
  }

  const observer = new IntersectionObserver((entries, obs) => {
    entries.forEach((entry) => {
      if (entry.isIntersecting) {
        http://entry.target.classList.add("is-visible");
        obs.unobserve(http://entry.target);
      }
    });
  }, { threshold: 0.14 });

  items.forEach((item) => observer.observe(item));
}

function boot() {
  applyStaticText();
  renderServices();
  renderSteps();
  initLanguage();
  initNavigation();
  initReveal();
}

## Отчёт по сайту

Создан статический рекламный сайт **«Klidný servis»** для выездного мастера компьютерных услуг. Концепция — «спокойная техническая помощь, которая возвращает человеку время»: тёплая палитра, мягкие карточки, крупная типографика и акцент на бесплатном выезде и первичной диагностике.

### Ключевые решения

- **Основной язык интерфейса:** чешский (`cs`).
- **Мультиязычность:** все пользовательские тексты вынесены в объект `I18N` в файле `js/app.js`; переключатель языка уже есть в шапке сайта.
- **Структура:** hero-блок, вводный блок, сетка из 12 услуг, блок процесса, учебный акцент, контактный блок.
- **Изображения:** все карточки используют пары `./img/slug--main.png` и `./img/slug--hover.png`; при наведении основное изображение плавно меняется на hover-версию.
- **Адаптивность:** сетка карточек перестраивается с 3–4 колонок на планшетах до одной колонки на мобильных.
- **Доступность:** есть skip-link, alt-тексты, фокус-состояния, семантические секции, `prefers-reduced-motion`.
- **Технологии:** чистые HTML, CSS и JavaScript без сборки и без серверной части.

### Палитра и стиль

- Основной тёмно-синий: `#16384f`.
- Дополнительный синий: `#245f78`.
- Тёплый фон: `#f7f2e9`.
- Акцент: `#f2a541`.
- Второй акцент: `#d9673f`.

Шрифты: **Manrope** для интерфейса и **Playfair Display** для крупного заголовка. Такое сочетание даёт современный, но человечный характер.

### Как менять язык

Сейчас активен словарь `cs`. Чтобы добавить второй язык:

1. В `js/app.js` добавьте новый ключ рядом с `cs`, например `ru` или `en`.
2. Повторите структуру вложенных полей.
3. Добавьте `<option value="ru">Русский</option>` или другой язык в `index.html`.
4. Скрипт автоматически перерисует статические тексты, карточки и шаги процесса.

## Развёртывание на Vercel

1. Создайте новый проект на [Vercel](https://vercel.com/).
2. Выберите один из вариантов:
    - загрузите папку проекта через GitHub/GitLab/Bitbucket;
    - либо используйте Vercel CLI.
3. Для Git-проекта:
    - загрузите все файлы в репозиторий;
    - подключите репозиторий в Vercel;
    - build command оставьте пустой или используйте `npm run check`;
    - output directory оставьте `.`.
4. Для Vercel CLI:
    
    ```
    npm i -g vercel
    vercel
    vercel --prod
    ```
    
5. После публикации проверьте:
    - открывается ли главная страница;
    - загружаются ли изображения из `img/`;
    - работает ли мобильное меню;
    - меняются ли изображения карточек при наведении;
    - открываются ли `mailto:`, `tel:` и WhatsApp-ссылки;
    - нет ли ошибок в консоли браузера.
