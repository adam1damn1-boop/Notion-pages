(function () {
	"use strict";

	const LANG_KEY = "marta-lang";
	const DEFAULT_LANG = "cs";

	/ ---------- i18n ---------- /
	function applyLang(lang) {
		const dict = (window.I18N && window.I18N[lang]) || window.I18N[DEFAULT_LANG];
		if (!dict) return;
		document.documentElement.lang = lang;
		document.querySelectorAll("[data-i18n]").forEach(function (el) {
			const key = el.getAttribute("data-i18n");
			if (dict[key] != null) {
				if (el.tagName === "TITLE") {
					document.title = dict[key];
				} else {
					el.textContent = dict[key];
				}
			}
		});
	}

	function initLangSwitch() {
		const sel = document.getElementById("lang-switch");
		if (!sel) return;
		// Populate from available dictionaries
		const available = Object.keys(window.I18N || {});
		if (available.length > 1) {
			sel.innerHTML = "";
			available.forEach(function (code) {
				const opt = document.createElement("option");
				opt.value = code;
				opt.textContent = code.toUpperCase();
				sel.appendChild(opt);
			});
		}
		let saved = DEFAULT_LANG;
		try { saved = localStorage.getItem(LANG_KEY) || DEFAULT_LANG; } catch () {}
		if (!window.I18N[saved]) saved = DEFAULT_LANG;
		sel.value = saved;
		applyLang(saved);
		sel.addEventListener("change", function () {
			applyLang(sel.value);
			try { localStorage.setItem(LANG_KEY, sel.value); } catch () {}
		});
	}

	/ ---------- Mobile nav ---------- /
	function initNav() {
		const btn = document.querySelector(".nav-toggle");
		const list = document.getElementById("nav-list");
		if (!btn || !list) return;
		btn.addEventListener("click", function () {
			const open = btn.getAttribute("aria-expanded") === "true";
			btn.setAttribute("aria-expanded", String(!open));
		});
		list.addEventListener("click", function (e) {
			if (http://e.target.tagName === "A") btn.setAttribute("aria-expanded", "false");
		});
	}

	/ ---------- Reveal on scroll ---------- /
	function initReveal() {
		const items = document.querySelectorAll("[data-reveal]");
		if (!("IntersectionObserver" in window) || !items.length) {
			items.forEach(function (el) { el.classList.add("is-visible"); });
			return;
		}
		const io = new IntersectionObserver(function (entries) {
			entries.forEach(function (entry) {
				if (entry.isIntersecting) {
					http://entry.target.classList.add("is-visible");
					io.unobserve(http://entry.target);
				}
			});
		}, { rootMargin: "0px 0px -10% 0px", threshold: 0.08 });
		items.forEach(function (el) { io.observe(el); });
	}

	/ ---------- Contact form -> mailto ---------- /
	function initForm() {
		const form = document.getElementById("contact-form");
		if (!form) return;
		form.addEventListener("submit", function (e) {
			e.preventDefault();
			const data = new FormData(form);
			const name = (data.get("name") || "").toString().trim();
			const contact = (data.get("contact") || "").toString().trim();
			const message = (data.get("message") || "").toString().trim();
			const subject = encodeURIComponent("Poptávka — " + (name || "Šálek čaje"));
			const body = encodeURIComponent(
				"Jméno: " + name + "n" +
				"Kontakt: " + contact + "nn" +
				message
			);
			window.location.href = "mailto:petr@salekcaje.cz?subject=" + subject + "&body=" + body;
		});
	}

	/ ---------- Footer year ---------- /
	function initYear() {
		const y = document.getElementById("year");
		if (y) y.textContent = String(new Date().getFullYear());
	}

	document.addEventListener("DOMContentLoaded", function () {
		initLangSwitch();
		initNav();
		initReveal();
		initForm();
		initYear();
	});
})();

## 📋 Отчёт по сайту

### Концепция

Макет «Šálek čaje» («Чашка чая») построен вокруг центральной метафоры исходных текстов: мастер заходит к вам в гости, ставится чайник, и все компьютерные проблемы решаются в спокойной домашней обстановке. Дизайн передаёт уют, доверие, теплоту и уверенность.

### Целевые устройства

Mobile-first адаптивная вёрстка с breakpoints: 480 px, 768 px, 1024 px, 1280 px. Корректно отображается на смартфонах, планшетах, ноутбуках и больших экранах.

### Палитра

- `#FBF7F0` — кремовый фон (молочный чай)
- `#E8DCC8` — мягкий бежевый (печенье к чаю)
- `#B8763A` — янтарно-чайный (заваренный чай)
- `#6B4423` — глубокий тёмный (крепкая заварка)
- `#7A8B6A` — приглушённая зелень (чайный лист)
- `#2A1F17` — тёмно-коричневый текст
- `#6F5D4D` — приглушённый коричневый текст

### Шрифты

- **Fraunces** (Google Fonts) — заголовки. Тёплая контрастная антиква, добавляет «домашний» характер.
- **Inter** (Google Fonts) — основной текст. Чистый, отлично читается на всех устройствах.

### Ключевые приёмы дизайна

- Hero-секция с большим иллюстрированным мастером (`hero--main.png` / `hero--hover.png`) и крупным слоганом из записи `main`.
- Карточки услуг с эффектом «листания»: при наведении плавно меняется изображение `slug--main.png` → `slug--hover.png`, появляется лёгкая тень и приподнятие.
- Группировка 12 услуг в 4 тематические категории (Péče o počítač / Bezpečnost / Vzdělávání / Design) с цветными заголовками-разделителями.
- Плавные scroll-анимации появления карточек через `IntersectionObserver`.
- Якорная навигация со smooth-scroll.
- Контактный блок с CTA-кнопками: `mailto:`, телефон (`tel:`), мессенджеры (WhatsApp, Telegram), плюс HTML5-форма (отправка через `mailto:`).
- Переключатель языков (по умолчанию доступен `cs`; структура подготовлена для добавления `en`, `ru` и других).
- Доступность: alt-тексты, контрастные пары, focus-стили, `prefers-reduced-motion`.
- Lazy-loading у всех изображений карточек.

### Обоснование решений

- **Тёплая цветовая гамма** прямо отражает тон текстов (чай, домашний уют).
- **Серифные заголовки** добавляют «человечности» и контрастируют с современным sans-serif для тела.
- **Карточная сетка** позволяет компактно показать все 12 услуг и работает на любых экранах.
- **Hover-смена изображения** — обязательное требование, использован безшовный CSS-фейдинг.
- **i18n-словарь** вынесен в отдельный JS-файл, поэтому добавление языка сводится к расширению одного объекта.

### Куда положить файлы из свойств записи

- **Html** → `index.html` в корне проекта.
- **Styles** → `styles/main.css`.
- **Scripts** → разделить по маркерам `=== … ===` на `scripts/i18n.js` и `scripts/main.js`.
- **Other** → разделить по маркерам `=== … ===` на `vercel.json`, `manifest.webmanifest`, `robots.txt`, `favicon.svg`.

### Папка `./img/`

1. Создайте папку `img` в корне.
2. Скачайте каждую пару изображений со страницы [Main images](https://www.notion.so/Main-images-3614fe82e7d4800891eacef2c9d4ab58?pvs=21).
3. Переименуйте файлы согласно правилам: `<slug>--main.png` и `<slug>--hover.png` (см. список выше).
4. Все ссылки в HTML/CSS уже указывают на `./img/<slug>--main.png` и `./img/<slug>--hover.png`.

### Как переключать языки

- Словарь находится в `scripts/i18n.js` в объекте `I18N`.
- По умолчанию подгружается `cs` (загружается из словаря в `DOMContentLoaded`).
- Чтобы добавить новый язык, скопируйте объект `cs`, переведите значения и положите рядом, например `en: { … }`.
- В переключателе языков в `index.html` (`<select id="lang-switch">`) добавьте `<option value="en">EN</option>`.
- Выбор пользователя сохраняется в `localStorage` (`marta-lang`).

## 🚀 Развёртывание на Vercel (или аналоге)

### Вариант A — через Git (рекомендуется)

1. Создайте репозиторий на GitHub/GitLab/Bitbucket и загрузите туда содержимое папки `site/`.
2. Зарегистрируйтесь на [vercel.com](http://vercel.com) и нажмите **Add New → Project**.
3. Подключите репозиторий. Vercel автоматически определит статический сайт.
4. В разделе **Build & Output Settings** оставьте всё по умолчанию (Output Directory = `./`).
5. Нажмите **Deploy**. Через 30–60 секунд сайт будет доступен на `your-project.vercel.app`.
6. В **Project Settings → Domains** можно подключить собственный домен.

### Вариант B — drag-and-drop

1. Сожмите папку `site/` в ZIP.
2. На главной Vercel воспользуйтесь **Upload** (или Vercel CLI: `npx vercel`).
3. После загрузки Vercel сразу выдаст URL.

### Netlify / Cloudflare Pages / GitHub Pages

Действия аналогичны: подключите репозиторий или загрузите папку. Никакой сборки не требуется — это чистая статика.

### Проверка домена

1. После деплоя откройте URL и убедитесь, что:
    - открывается главная страница;
    - изображения подгружаются (если нет — проверьте регистр имён файлов в `./img/`);
    - переключатель языков работает;
    - формы открывают почтового клиента.
2. Прогоните страницу через [PageSpeed Insights](https://pagespeed.web.dev/) — для статической версии ожидаем 95+ Performance.
3. Прогоните валидацию HTML — [validator.w3.org](http://validator.w3.org).
