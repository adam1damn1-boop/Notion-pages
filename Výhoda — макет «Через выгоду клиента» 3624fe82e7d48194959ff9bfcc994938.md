(function () {
	'use strict';

	var DEFAULT_LANG = 'cs';
	var STORAGE_KEY = 'vyhoda.lang';
	var dict = window.I18N || {};

	function getLang() {
		try {
			var saved = localStorage.getItem(STORAGE_KEY);
			if (saved && dict[saved]) return saved;
		} catch (e) {}
		return DEFAULT_LANG;
	}

	function applyLang(lang) {
		if (!dict[lang]) lang = DEFAULT_LANG;
		var pack = dict[lang];
		document.documentElement.setAttribute('lang', lang);
		document.documentElement.setAttribute('data-lang', lang);
		var nodes = document.querySelectorAll('[data-i18n]');
		nodes.forEach(function (node) {
			var key = node.getAttribute('data-i18n');
			if (!key || !(key in pack)) return;
			var val = pack[key];
			if (node.tagName === 'META') {
				node.setAttribute('content', val);
			} else if (node.tagName === 'TITLE') {
				node.textContent = val;
			} else if (node.tagName === 'INPUT' && node.type === 'submit') {
				node.value = val;
			} else {
				node.textContent = val;
			}
		});
		document.querySelectorAll('[data-lang-switch]').forEach(function (b) {
			b.classList.toggle('is-active', b.getAttribute('data-lang-switch') === lang);
		});
		try { localStorage.setItem(STORAGE_KEY, lang); } catch (e) {}
	}

	function initLangSwitch() {
		document.querySelectorAll('[data-lang-switch]').forEach(function (b) {
			b.addEventListener('click', function () {
				applyLang(b.getAttribute('data-lang-switch'));
			});
		});
	}

	function initBurger() {
		var burger = document.querySelector('.burger');
		var nav = document.querySelector('.nav__links');
		if (!burger || !nav) return;
		burger.addEventListener('click', function () {
			var open = nav.classList.toggle('is-open');
			burger.setAttribute('aria-expanded', open ? 'true' : 'false');
		});
		nav.querySelectorAll('a').forEach(function (a) {
			a.addEventListener('click', function () {
				nav.classList.remove('is-open');
				burger.setAttribute('aria-expanded', 'false');
			});
		});
	}

	function initScrollHeader() {
		var header = document.querySelector('.site-header');
		if (!header) return;
		var onScroll = function () {
			header.classList.toggle('is-scrolled', window.scrollY > 8);
		};
		document.addEventListener('scroll', onScroll, { passive: true });
		onScroll();
	}

	function initReveal() {
		var items = document.querySelectorAll('.section, .card, .callout, .contact__form');
		items.forEach(function (el) { el.classList.add('reveal'); });
		if (!('IntersectionObserver' in window)) {
			items.forEach(function (el) { el.classList.add('is-visible'); });
			return;
		}
		var io = new IntersectionObserver(function (entries) {
			entries.forEach(function (e) {
				if (e.isIntersecting) {
					http://e.target.classList.add('is-visible');
					io.unobserve(http://e.target);
				}
			});
		}, { threshold: 0.12, rootMargin: '0px 0px -40px 0px' });
		items.forEach(function (el) { io.observe(el); });
	}

	function initTouchSwap() {
		if (!('ontouchstart' in window)) return;
		document.querySelectorAll('.swap').forEach(function (el) {
			el.addEventListener('touchstart', function () {
				document.querySelectorAll('.http://swap.is-touched').forEach(function (o) { if (o !== el) o.classList.remove('is-touched'); });
				el.classList.toggle('is-touched');
			}, { passive: true });
		});
	}

	function initYear() {
		var y = document.getElementById('year');
		if (y) y.textContent = new Date().getFullYear();
	}

	document.addEventListener('DOMContentLoaded', function () {
		applyLang(getLang());
		initLangSwitch();
		initBurger();
		initScrollHeader();
		initReveal();
		initTouchSwap();
		initYear();
	});
})();

# Отчёт по сайту «Výhoda»

## Концепция

**Через выгоду клиента.** Главный посыл сайта построен на формуле «вы платите малую часть — получаете полную скорость / спокойствие / уверенность». Каждая карточка услуги напоминает о бесплатной диагностике и выезде, без давления. Имя бренда — **Výhoda** (чеш. *выгода*).

## Целевые устройства

- Mobile-first: тестировался брейкпоинт ≤ 720px (одна колонка, бургер-меню, скрытый CTA-кнопка в шапке).
- Планшет ≤ 1024px: 2-колоночная сетка карточек, hero и design-секция перестраиваются в один столбец.
- Desktop ≥ 1025px: 4 карточки в ряд для «Péče», 3 в ряд для «Bezpečnost» и «Učení», hero — 2 колонки, контакт — 2 колонки.

## Ключевые приёмы дизайна

- **Палитра** — глубокий навигационный (`#0f2540`), тёплый кремовый фон (`#fbf8f1`), акцентное золото (`#d99a2b`) и приглушённый изумруд (`#16624c`). Это создаёт ощущение надёжности и тепла.
- **Типографика** — *Spectral* (заголовки, акценты — итальянский с курсивом) + *Manrope* (UI, тексты). Контраст serif/sans-serif создаёт «премиальное» ощущение.
- **Сетка** — CSS Grid + Flexbox, контейнер 1180px, отступы 24px.
- **Микровзаимодействия** — hover-смена изображений `swap__main` → `swap__hover` для всех 12 карточек, hero и design-арта; мягкий подъём карточки `translateY(-6px)`, увеличение фото 1.04×, плавная подсветка под навигацией. На touch-устройствах эффект работает по тапу (class `is-touched`).
- **Анимации** — IntersectionObserver реализует мягкое появление секций; маркетинговая «бегущая строка» под hero. Все анимации уважают `prefers-reduced-motion`.
- **Декор** — мягкие радиальные градиенты на hero, светящаяся «бегущая строка» как разделитель, тёплый блок callout с золотой обводкой.
- **Доступность** — alt-тексты, skip-link, фокус-стили на инпутах, контраст AA, ARIA-атрибуты для бургера и переключателя языка.

## Обоснование решений

- Тёплая палитра вместо «техно-холодной» — потому что клиент мастера платит за **спокойствие и доверие**, а не за гик-стиль.
- Один большой блок «Pro dospělé» (карточка-баннер с изображением literacy) — выделяет курс для взрослых, который часто кажется им неудобной темой.
- Section «Bezpečnost» затемнена (фон бренда) — психологически передаёт «защиту».
- Callout с пятиконечной звездой повторяет авторитетный тон оригинальной страницы услуг.

## Как разрезать свойства Notion на файлы

1. Открыть запись Site code → каждое из свойств Html / Styles / Scripts / Other.
2. Внутри значения свойств блоки разделены маркерами вида `=== filename ===` — взять текст между двумя такими маркерами и сохранить в файл с указанным именем.
3. Изображения положить в папку `./img/` строго с именами `slug--main.png` и `slug--hover.png`, где `slug` — слаг из страницы Main images.

## Как переключать языки

- Все строки лежат в `i18n.js` в объекте `window.I18N`. Язык по умолчанию — `cs`.
- В DOM-узлах используется атрибут `data-i18n="ключ"`. Скрипт `scripts.js` (`applyLang`) подменяет `textContent` или `content` для `<meta>`.
- Чтобы добавить язык, создайте новую ветку в `I18N` (например, `de: { ... }`) и кнопку `<button data-lang-switch="de">DE</button>` в `.lang`.
- Выбранный язык запоминается в `localStorage` (`vyhoda.lang`).

# Развёртывание на бесплатном хостинге

## Вариант 1 — Vercel (рекомендовано)

1. Создайте репозиторий на GitHub и загрузите туда все файлы из структуры выше.
2. Зайдите на [vercel.com](http://vercel.com), нажмите **Add New… → Project** и подключите репозиторий.
3. Framework Preset: **Other** (это статический сайт). Команды Build / Output Directory оставьте пустыми — Vercel сам отдаст файлы из корня. Файл `vercel.json` подключит правильные заголовки и кеш.
4. Нажмите **Deploy**. Через ~30 секунд получите домен вида `vyhoda.vercel.app`.
5. В разделе **Domains** добавьте собственный домен (например, `vyhoda.cz`) и пропишите DNS-записи, которые покажет Vercel.
6. После HTTPS-сертификата сайт готов.

## Вариант 2 — Netlify

1. На [netlify.com](http://netlify.com) выберите **Add new site → Deploy manually** и перетащите папку проекта.
2. Готово — Netlify выдаст ссылку. Для своего домена — раздел Domain settings.

## Вариант 3 — GitHub Pages

1. В репозитории включите Pages: Settings → Pages → Source: `Deploy from a branch` → `main` / root.
2. Сайт будет доступен на `https://<user>.github.io/<repo>/`.
3. Для собственного домена добавьте файл `CNAME` с доменом и пропишите DNS.

## Проверка после деплоя

- Откройте сайт в Chrome и Safari, проверьте Lighthouse (Performance, Accessibility, Best Practices ≥ 90).
- Проверьте мобильную версию в DevTools (iPhone SE, Pixel 7).
- Кликните по всем якорным ссылкам, переключите язык, отправьте тестовое сообщение через форму (FormSubmit потребует подтверждения e-mail).
- Убедитесь, что hover-смена изображений работает на всех 12 карточках, hero и design-блоке.
