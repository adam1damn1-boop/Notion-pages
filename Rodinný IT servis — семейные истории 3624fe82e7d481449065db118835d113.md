const getLang = () => localStorage.getItem("site-lang") || "cs";
const t = (path) => path.split(".").reduce((acc, part) => acc && acc[part], I18N[getLang()]) || path;

function applyI18n(){
  document.documentElement.lang = getLang();
  document.querySelectorAll("[data-i18n]").forEach(el => { el.textContent = t(el.dataset.i18n); });
}

function renderGrid(){
  const grid = document.querySelector("[data-service-grid]");
  if(!grid) return;
  grid.innerHTML = http://SERVICES.map((svc, index) => {
    const item = I18N[getLang()].services[svc.key];
    return <a class="service-card reveal" href="./${svc.file}" style="animation-delay:${Math.min(index*0.04,.35)}s">
      <div class="service-card__image">
        <img class="main" src="./img/${svc.key}--main.png" alt="${item.alt}" loading="lazy">
        <img class="hover" src="./img/${svc.key}--hover.png" alt="" loading="lazy" aria-hidden="true">
      </div>
      <div class="service-card__body">
        <span class="service-card__meta">${svc.section}</span>
        <h3>${item.title}</h3>
        <p>${item.subtitle}</p>
      </div>
    </a>;
  }).join("");
}

function renderServicePage(){
  const mount = document.querySelector("[data-service-page]");
  if(!mount) return;
  const key = document.body.dataset.service;
  const index = SERVICES.findIndex(s => s.key === key);
  const svc = SERVICES[index];
  const item = I18N[getLang()].services[key];
  const prev = SERVICES[(index - 1 + SERVICES.length) % SERVICES.length];
  const next = SERVICES[(index + 1) % SERVICES.length];
  mount.innerHTML = 
    <section class="hero service-hero">
      <div class="hero__text reveal">
        <p class="eyebrow">${svc.section} · ${t("common.free")}</p>
        <h1>${item.title}</h1>
        <p class="lead">${item.subtitle}</p>
        <div class="hero__actions"><a class="button" href="./kontakt.html">${t("common.cta")}</a><a class="button button--soft" href="./index.html">${t("common.backHome")}</a></div>
      </div>
      <figure class="hero-card reveal">
        <img src="./img/${key}--main.png" alt="${item.alt}" loading="eager">
        <figcaption>${item.subtitle}</figcaption>
      </figure>
    </section>
    <section class="section service-body">
      <article class="story reveal">
        <p>${item.text}</p>
        <ul class="feature-list">${item.points.map(point => <li>${point}</li>).join("")}</ul>
        <nav class="page-nav" aria-label="Navigace mezi službami">
          <a class="button button--soft" href="./${prev.file}">← ${t("common.previous")}</a>
          <a class="button" href="./${next.file}">${t("common.next")} →</a>
        </nav>
      </article>
      <figure class="side-image reveal">
        <img src="./img/${key}--hover.png" alt="${item.alt} – detail" loading="lazy">
      </figure>
    </section>;
}

function initLanguage(){
  const select = document.querySelector("[data-language]");
  if(!select) return;
  select.value = getLang();
  select.addEventListener("change", () => {
    localStorage.setItem("site-lang", select.value);
    applyI18n(); renderGrid(); renderServicePage();
  });
}

document.addEventListener("DOMContentLoaded", () => {
  initLanguage();
  applyI18n();
  renderGrid();
  renderServicePage();
});

## Отчёт по сайту

Создан макет **«Rodinný IT servis»** для записи Site content **«Семейные истории»**.

### Ассоциации из брифа и текстов

- **Домашнее тепло** — услуги поданы через кухонный стол, бабушку, детей, семейные фото и бытовые сценки.
- **Забота без давления** — мастер сначала объясняет, что происходит, и только потом предлагает решение.
- **Надёжный соседский помощник** — выезд и диагностика бесплатны, тон спокойный и понятный.
- **Техника как часть семьи** — компьютер, камеры, сигнализация и AI помогают людям, а не выглядят холодной технологией.
- **Мягкая безопасность** — защита дома и данных без агрессивной «кибер»-стилистики.

Финальный макет этим ассоциациям соответствует: тёплый фон, бумажные карточки, мягкие тени, округлые формы, семейная типографика, спокойные микроанимации и понятная навигация.

### Концепция

Сайт сделан как доброжелательный семейный сервис: не «ремонтная мастерская», а человек, который приходит домой и спокойно решает технические проблемы. Главная страница показывает все 12 услуг карточками с hover-сменой изображений. Каждая услуга вынесена на отдельную HTML-страницу с собственной обложкой, текстом, списком пользы и соседней навигацией.

### Устройства и адаптивность

- Mobile-first сетка с перестроением карточек на мобильных экранах.
- Sticky-шапка с прозрачным размытием.
- Крупные кликабельные кнопки и ссылки.
- Учитывается `prefers-reduced-motion`.

### Палитра и типографика

- Основной фон: тёплый кремовый `#fff8ed`.
- Текст: глубокий коричневый `#2f241d`.
- Акцент: терракотовый `#c45f3d`.
- Дополнительные оттенки: шалфейный, небесный, масляно-жёлтый.
- Шрифты: **Fraunces** для заголовков и **Nunito** для интерфейса и основного текста.

### Проверка структуры

- Есть `index.html`.
- Есть 12 отдельных страниц услуг.
- Есть `kontakt.html`.
- Все внутренние ссылки относительные.
- Все изображения подключаются только из `./img/` по схеме `slug--main.png` и `slug--hover.png`.
- Hover-смена изображения используется только на карточках главной страницы.
- На страницах услуг `--main` используется как hero, `--hover` — как контекстная иллюстрация.
- Словарь `i18n.js` содержит тексты `cs` и готов к расширению другими языками.
- Контактные телефон и e-mail добавлены как заменяемые заглушки, потому что конкретные контактные данные в источнике не указаны.

## Инструкция по сборке

1. Создайте папку проекта.
2. Скопируйте файлы из свойства **Html** в корень проекта:
    - `index.html`
    - 12 страниц услуг
    - `kontakt.html`
3. Скопируйте файл из свойства **Styles**:
    - `styles.css`
4. Скопируйте файл из свойства **Scripts**:
    - `i18n.js`
5. Создайте папку `img`.
6. Поместите в `img` изображения со страницы Main images:
    - `hero--main.png`, `hero--hover.png`
    - `cleaning--main.png`, `cleaning--hover.png`
    - `upgrade--main.png`, `upgrade--hover.png`
    - `software--main.png`, `software--hover.png`
    - `repair--main.png`, `repair--hover.png`
    - `webcam--main.png`, `webcam--hover.png`
    - `alarm--main.png`, `alarm--hover.png`
    - `appliances--main.png`, `appliances--hover.png`
    - `literacy--main.png`, `literacy--hover.png`
    - `tutor--main.png`, `tutor--hover.png`
    - `chess--main.png`, `chess--hover.png`
    - `development--main.png`, `development--hover.png`
    - `design--main.png`, `design--hover.png`
7. Замените в `kontakt.html` и `README.md` контактные заглушки:
    - `+420 000 000 000`
    - `servis@example.cz`
    - ссылку WhatsApp.
8. Откройте `index.html` в браузере или запустите локальный сервер.

## Переключение языков

Сейчас активен язык `cs`. Для добавления другого языка нужно:

1. Добавить новый ключ верхнего уровня в объект `I18N`, например `ru` или `en`.
2. Повторить структуру `nav`, `home`, `contact`, `services`.
3. Добавить новый `<option>` в переключатель языка на всех HTML-страницах.

## Развёртывание на Vercel

1. Создайте новый Git-репозиторий или подготовьте папку проекта вручную.
2. Проверьте, что в корне проекта лежат HTML-файлы, `styles.css`, `i18n.js`, `vercel.json`, `package.json`, а изображения — в папке `img`.
3. Зайдите на Vercel и создайте новый проект.
4. Если используете Git — подключите репозиторий. Если без Git — загрузите папку через интерфейс или Vercel CLI.
5. Build command оставьте пустой: сайт статический и не требует сборки.
6. Output directory оставьте корнем проекта.
7. После деплоя откройте `index.html`, проверьте переходы на все 12 страниц и страницу контакта.
8. Проверьте, что изображения загружаются, hover на главной работает, а формы открывают почтовый клиент.
